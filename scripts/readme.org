#+title: Voice of the Customer Analysis Scripts

This folder contains examples in the R language for statistical computing of how to extract he Voice of the Customer from raw data using the case studies.

#+begin_src bash :results raw
  # List files and extract first comment line
  for f in *.[Rr]; do
    [ -f "$f" ] || continue
    read -r line < "$f"
    echo "- =$f=: $(echo "$line" | sed 's/^#*[[:space:]]*//')"
  done
#+end_src
